🛠️ Установка сервера за 5 минут

# 1. Клонируйте репозиторий
git clone https://github.com/olegmmg/telegramwhite-server
cd telegramwhite-server

# 2. Установите зависимости
pip install -r requirements.txt

# 3. Настройте базу данных
# Создайте PostgreSQL базу данных и укажите DATABASE_URL

# 4. Запустите сервер
python server.py
