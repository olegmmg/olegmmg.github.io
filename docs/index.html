💻 Готовый код

JavaScript

const SERVERS = [
  { url: "https://radar-u913.onrender.com", key: "КЛЮЧ_ОСН" },
  { url: "https://radar-twp9.onrender.com", key: "КЛЮЧ_РЕЗ" }
];
async function getStatuses() {
  for (const s of SERVERS) {
    try {
      const r = await fetch(`${s.url}/api/statuses?api_key=${s.key}`);
      if (r.ok) return await r.json();
    } catch (e) {}
  }
  throw new Error("Серверы недоступны");
}

Python

import requests
SERVERS = [
  {"url":"https://radar-u913.onrender.com","key":"КЛЮЧ_ОСН"},
  {"url":"https://radar-twp9.onrender.com","key":"КЛЮЧ_РЕЗ"}
]
def get_statuses():
  for s in SERVERS:
    try:
      r = requests.get(f"{s['url']}/api/statuses", params={"api_key":s['key']}, timeout=10)
      if r.status_code == 200: return r.json()
    except: continue
  raise Exception("Серверы недоступны")
